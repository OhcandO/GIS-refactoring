예제 목록 오버레이 (팝업) API 문서

지도 위에 드래그 가능한 오버레이 팝업을 표시합니다. MOOverlay는 OpenLayers Overlay를 확장하여 드래그 기능을 내장합니다. 지도를 클릭하면 해당 위치에 팝업이 생성됩니다.

import { createBaseMap, initCodeToggle } from './js/common.js';
import { MOOverlay } from './js/mogismap.js';

const mogisMap = createBaseMap('map');
let overlayCount = 0;

// 맵 클릭 시 팝업 생성
mogisMap.map.on('singleclick', (evt) => {
  overlayCount++;

  const el = document.createElement('div');
  el.className = 'popup';
  el.innerHTML = `
    <div class="popup-header">팝업 #${overlayCount}</div>
    <div class="popup-content">
      좌표: ${evt.coordinate[0].toFixed(0)}, ${evt.coordinate[1].toFixed(0)}<br>
      드래그하여 이동할 수 있습니다.
    </div>
  `;
  document.body.appendChild(el);

  const overlay = new MOOverlay({
    element: el,
    position: evt.coordinate,
    positioning: 'bottom-center',
    offset: [0, -10],
    id: `popup-${overlayCount}`,
  });

  mogisMap.addMOverlay(overlay);
});

// 오버레이 전체 보이기/숨기기
let visible = true;
document.getElementById('toggle-overlays').addEventListener('click', () => {
  visible = !visible;
  mogisMap.ctrlOverlay(undefined, visible);
  document.getElementById('toggle-overlays').textContent = visible ? '팝업 숨기기' : '팝업 보이기';
});

// 오버레이 전체 제거
document.getElementById('clear-overlays').addEventListener('click', () => {
  mogisMap.discardMOverlay();
  overlayCount = 0;
});
