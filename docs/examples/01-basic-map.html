예제 목록 기본 지도 생성 API 문서

MOGISMap의 가장 기본적인 사용법입니다. 인스턴스를 생성하고 배경지도(OSM)를 표출합니다.

import { createBaseMap, initCodeToggle } from './js/common.js';

// 1. MOGISMap 인스턴스 생성 (OSM 배경지도 자동 적용)
const mogisMap = createBaseMap('map');

// 2. 뷰 조정 예시
document.getElementById('btn-seoul').addEventListener('click', () => {
  mogisMap.view.animate({ center: [14135490, 4518350], zoom: 14, duration: 1000 });
});
document.getElementById('btn-incheon').addEventListener('click', () => {
  mogisMap.view.animate({ center: [14102530, 4504570], zoom: 13, duration: 1000 });
});
