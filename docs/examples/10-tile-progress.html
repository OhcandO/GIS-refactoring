예제 목록 타일 로딩 프로그레스 API 문서

배경지도 타일이 로딩될 때 진행률 바를 표시합니다. 아래 버튼을 눌러 지도를 이동시키면 타일 로딩이 발생하면서 프로그레스 바가 동작합니다.

import { createBaseMap, initCodeToggle } from './js/common.js';
import { TileLoadProgress } from './js/mogismap.js';

const mogisMap = createBaseMap('map');
const progressEl = document.getElementById('progress');
const progress = new TileLoadProgress(progressEl);

// 배경지도 타일 소스에 이벤트 연결
const layers = mogisMap.map.getLayers().getArray();
layers.forEach(layer => {
  const source = layer.getSource();
  if (source && source.on) {
    source.on('tileloadstart', () => progress.addLoading());
    source.on('tileloadend', () => progress.addLoaded());
    source.on('tileloaderror', () => progress.addLoaded());
  }
});

// 지도 이동 버튼 (타일 로드 트리거)
document.getElementById('btn-move').addEventListener('click', () => {
  const view = mogisMap.view;
  const center = view.getCenter();
  view.animate({ center: [center[0] + 50000, center[1]], duration: 500 }, () => {
    view.animate({ center: center, duration: 500 });
  });
});

initCodeToggle();
