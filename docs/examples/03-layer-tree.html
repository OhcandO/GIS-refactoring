예제 목록 레이어 트리 API 문서

LayerTree를 사용하여 계층형 체크박스 UI로 레이어를 관리합니다. Publisher/Subscriber 패턴으로 트리와 지도가 연동됩니다.

레이어 트리

이 예제에서는 GeoServer 없이 레이어 코드만 등록하여 트리 UI를 구성합니다. 실제 운영 환경에서는 ctrlLayer가 SourceFactory를 통해 레이어를 자동 생성합니다. 
import { createBaseMap, initCodeToggle } from './js/common.js';
import { LayerTree, KEY } from './js/mogismap.js';

const mogisMap = createBaseMap('map');

// 계층형 레이어 코드 정의
const layerCodes = [
  { id: 100, pid: null, layerTitle: '상수도 시설', ordr: 1, boolIsgroup: 'Y', boolUseYn: 'Y', geomType: null },
  { id: 101, pid: 100, layerTitle: '제수밸브', ordr: 1, boolIsgroup: 'N', boolUseYn: 'Y', geomType: 'Point', typeName: 'valve', boolSelectable: 'Y', boolShowInit: 'N' },
  { id: 102, pid: 100, layerTitle: '상수관로', ordr: 2, boolIsgroup: 'N', boolUseYn: 'Y', geomType: 'LineString', typeName: 'pipe', boolSelectable: 'Y', boolShowInit: 'N' },
  { id: 200, pid: null, layerTitle: '관리 구역', ordr: 2, boolIsgroup: 'Y', boolUseYn: 'Y', geomType: null },
  { id: 201, pid: 200, layerTitle: '급수블록', ordr: 1, boolIsgroup: 'N', boolUseYn: 'Y', geomType: 'Polygon', typeName: 'block', boolSelectable: 'Y', boolShowInit: 'N' },
];

mogisMap.setLayerCode(layerCodes, 'base');

// LayerTree 생성 및 등록
const layerTree = new LayerTree('tree-container');
layerTree.regist(mogisMap, 'base');
