예제 목록 거리/면적 측정 API 문서

지도 위에 선이나 면을 그려 거리와 면적을 실시간으로 측정합니다.

import { createBaseMap, initCodeToggle } from './js/common.js';
import { MOMeasure } from './js/mogismap.js';

const mogisMap = createBaseMap('map');
window.mainMap = mogisMap; // MOMeasure 내부에서 mainMap 전역 참조

const measure = new MOMeasure(mogisMap, false);

document.getElementById('btn-distance').addEventListener('click', () => {
  measure.activeMeasure('LineString');
  setActiveBtn('btn-distance');
});
document.getElementById('btn-area').addEventListener('click', () => {
  measure.activeMeasure('Polygon');
  setActiveBtn('btn-area');
});
document.getElementById('btn-reset').addEventListener('click', () => {
  measure.activeMeasure(false);
  setActiveBtn(null);
});

function setActiveBtn(id) {
  ['btn-distance', 'btn-area', 'btn-reset'].forEach(btnId => {
    document.getElementById(btnId).classList.toggle('btn-active', btnId === id);
  });
}

initCodeToggle();
