예제 목록 실시간 스타일 편집 API 문서

LayerTreeColorPickr를 사용하여 벡터 레이어의 색상을 실시간으로 변경합니다. 트리 노드 옆의 색상 칩을 클릭하면 ColorPickr 팝업이 열리며, theme, defaultRepresentation 등의 Pickr 옵션을 주입할 수 있습니다.

레이어 트리

import { createBaseMap, initCodeToggle } from './js/common.js';
import { LayerTreeColorPickr, KEY } from './js/mogismap.js';

window.ctxPath = '';
const mogisMap = createBaseMap('map');

const layerCodes = [
  { id: 100, pid: null, layerTitle: '벡터 레이어', ordr: 1, boolIsgroup: 'Y', boolUseYn: 'Y', geomType: null },
  { id: 101, pid: 100, layerTitle: '상수관로', ordr: 1, boolIsgroup: 'N', boolUseYn: 'Y', geomType: 'LineString', typeName: 'pipe',
    colorLine: 'rgba(52,152,219,0.8)', lineWidth: 3, lineStyle: 'solid',
    boolSelectable: 'Y', boolShowInit: 'N', layerPurposeCategory: 'base' },
  { id: 102, pid: 100, layerTitle: '급수블록', ordr: 2, boolIsgroup: 'N', boolUseYn: 'Y', geomType: 'Polygon', typeName: 'block',
    colorLine: 'rgba(46,204,113,0.8)', colorFill: 'rgba(46,204,113,0.2)', lineWidth: 2,
    boolSelectable: 'Y', boolShowInit: 'N', layerPurposeCategory: 'base' },
];

mogisMap.setLayerCode(layerCodes, 'base');

const colorPickrTree = new LayerTreeColorPickr('tree-container', {
  theme: 'monolith',
  defaultRepresentation: 'RGBA',
});
colorPickrTree.regist(mogisMap, 'base');

initCodeToggle();
