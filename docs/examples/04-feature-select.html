import { createBaseMap, loadGeoJSON, initCodeToggle, VectorImageLayer, Style, Stroke, Fill, CircleStyle } from './js/common.js';

const mogisMap = createBaseMap('map', { hitTolerance: 10 });

async function init() {
  // 포인트 레이어 추가
  const pointSrc = await loadGeoJSON('./data/sample-points.geojson');
  const pointLayer = new VectorImageLayer({
    source: pointSrc,
    style: new Style({
      image: new CircleStyle({ radius: 8, fill: new Fill({ color: '#e74c3c' }), stroke: new Stroke({ color: '#fff', width: 2 }) })
    }),
    zIndex: 30,
  });
  mogisMap.map.addLayer(pointLayer);

  // 선택 활성화
  mogisMap.enableSelect(true);
  mogisMap.setSelectCallback((feature, layer) => {
    const props = feature.getProperties();
    const infoDiv = document.getElementById('info');
    if (!props.geometry) return;

    let html = '<table class="info-table"><thead><tr><th>속성</th><th>값</th></tr></thead><tbody>';
    for (const [key, value] of Object.entries(props)) {
      if (key === 'geometry') continue;
      html += '<tr><td>' + key + '</td><td>' + value + '</td></tr>';
    }
    html += '</tbody></table>';
    infoDiv.innerHTML = html;
  });

  // 선택 토글
  let selectEnabled = true;
  document.getElementById('toggle-select').addEventListener('click', () => {
    selectEnabled = !selectEnabled;
    mogisMap.enableSelect(selectEnabled);
    document.getElementById('toggle-select').textContent = selectEnabled ? '선택 비활성화' : '선택 활성화';
  });
}

init();
